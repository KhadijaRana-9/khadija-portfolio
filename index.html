// server/src/shared/utils/leadScoring.js — the actual file, not a mockup
const URGENCY_SCORES = { immediate: 25, '1-3m': 15, '3-6m': 8, exploring: 3 };

function scoreBudgetMatch(budget, price) {
  if (!budget || !price) return 0;
  const closeness = 1 - Math.abs(budget - price) / price;
  return Math.round(Math.max(0, closeness) * 30);
}

function scorePopularity(views) {
  if (!views || views <= 0) return 0;
  const scaled = Math.log(views + 1) / Math.log(20);
  return Math.round(Math.min(1, scaled) * 20);
}
// scoreInterest() omitted here for length — scores message length + a phone-number bonus

function calculateLeadScore({ budget, price, moveTimeline, message, phone, propertyViews }) {
  const total = Math.min(100,
    scoreBudgetMatch(budget, price) +
    (URGENCY_SCORES[moveTimeline] ?? 0) +
    scoreInterest(message, phone) +
    scorePopularity(propertyViews));

  return { total, status: total >= 70 ? 'hot' : total >= 40 ? 'warm' : 'cold' };
}
